cmake_minimum_required(VERSION 3.16)
project(hmi-vms VERSION 1.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

option(HMI_CPU_TUNING "Tune code for the RK3588 CPU cores (Cortex-A76/A55) on aarch64" ON)
option(HMI_LTO "Enable link-time optimisation" ON)

find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Multimedia Network Widgets)
# Optional: Flussonic MSE-LD (fMP4 over WebSocket) playback.
find_package(Qt6 6.4 QUIET OPTIONAL_COMPONENTS WebSockets)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET
    gstreamer-1.0>=1.20 gstreamer-video-1.0>=1.20 gstreamer-app-1.0>=1.20
    gstreamer-pbutils-1.0>=1.20)

qt_standard_project_setup()

qt_add_executable(hmi-vms
    src/main.cpp
    src/gstutils.h src/gstutils.cpp
    src/player.h src/player.cpp
    src/streamprofile.h src/streamprofile.cpp
    src/camai.h src/camai.cpp
    src/discovery.h src/discovery.cpp
    src/vmswindow.h src/vmswindow.cpp
    src/appcontroller.h src/appcontroller.cpp
    src/diagnostics.h src/diagnostics.cpp
    src/branding.h src/branding.cpp
)

# QML sources live in qml/ but are placed at the module root (no extra
# directory -> no qmldir/QTP0004 handling needed).
set(HMI_QML_FILES
    qml/MonitorWindow.qml
    qml/PlayerTile.qml
    qml/PlayerDialog.qml
    qml/WindowDialog.qml
    qml/ToolIconButton.qml
    qml/HeaderButton.qml
    qml/ConfirmDialog.qml
    qml/DiagnosticsWindow.qml
    qml/SettingsDialog.qml
    qml/Icon.qml
)
foreach(f IN LISTS HMI_QML_FILES)
    get_filename_component(n "${f}" NAME)
    set_source_files_properties("${f}" PROPERTIES QT_RESOURCE_ALIAS "${n}")
endforeach()

qt_add_qml_module(hmi-vms
    URI HmiUi
    VERSION 1.0
    RESOURCE_PREFIX /
    NO_IMPORT_SCAN               # shared Qt: QML plugins are loaded at run time
    QML_FILES ${HMI_QML_FILES}
)

# Built-in wallpapers (generated by tools/make-wallpapers.py).
qt_add_resources(hmi-vms "wallpapers"
    PREFIX /wallpapers
    BASE resources/wallpapers
    FILES
        resources/wallpapers/camai-grid.jpg
        resources/wallpapers/camai-lens.jpg
        resources/wallpapers/camai-network.jpg
        resources/wallpapers/camai-hex.jpg
        resources/wallpapers/camai-waves.jpg
        resources/wallpapers/camai-graphite.jpg
)

        resources/branding/app-light-128.png

target_compile_definitions(hmi-vms PRIVATE
    HMI_VERSION="${PROJECT_VERSION}"
    QT_NO_KEYWORDS            # GStreamer/GLib headers use "signals"
    $<$<CONFIG:Release>:QT_NO_DEBUG_OUTPUT>)

target_link_libraries(hmi-vms PRIVATE
    Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2
    Qt6::Multimedia Qt6::Network Qt6::Widgets
    PkgConfig::GST)

if(HMI_CPU_TUNING AND CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
    include(CheckCXXCompilerFlag)
    check_cxx_compiler_flag("-mcpu=cortex-a76.cortex-a55" HMI_HAS_BIGLITTLE)
    if(HMI_HAS_BIGLITTLE)
        target_compile_options(hmi-vms PRIVATE -mcpu=cortex-a76.cortex-a55)
    endif()
endif()

if(HMI_LTO)
    include(CheckIPOSupported)
    check_ipo_supported(RESULT HMI_IPO OUTPUT HMI_IPO_MSG)
    if(HMI_IPO)
        set_property(TARGET hmi-vms PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
    endif()
endif()

if(TARGET Qt6::WebSockets)
    target_link_libraries(hmi-vms PRIVATE Qt6::WebSockets)
    target_compile_definitions(hmi-vms PRIVATE HMI_HAVE_WEBSOCKETS)
    message(STATUS "MSE-LD (WebSocket) playback: enabled")
else()
    message(WARNING "Qt6 WebSockets not found (qt6-websockets-dev): MSE-LD / ws:// playback disabled")
endif()

include(GNUInstallDirs)
install(TARGETS hmi-vms RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/hmi-vms.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
# Menu icon: the variant for light themes system-wide; while running, the
# program puts the variant for the current theme into ~/.local/share/icons.
foreach(s 16 22 24 32 48 64 128 256 512)
    install(FILES resources/branding/app-light-${s}.png
            DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/${s}x${s}/apps RENAME hmi-vms.png)
endforeach()
install(FILES resources/branding/app-light.svg
        DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps RENAME hmi-vms.svg)
